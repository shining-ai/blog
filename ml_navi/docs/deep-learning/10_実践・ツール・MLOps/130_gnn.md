# グラフニューラルネットワーク（GNN）

## GNN とは

> グラフニューラルネットワーク（Graph Neural Network, GNN）とは、グラフ構造データに対してニューラルネットワークを適用する手法の総称。ノード・エッジ・グラフ全体の表現を学習し、分子設計・ソーシャルネットワーク分析・知識グラフなどに応用される。

| 手法 | 特徴 | 主な用途 |
|------|------|----------|
| GCN（Graph Convolutional Network）| スペクトル法ベースの畳み込み | ノード分類 |
| GraphSAGE | 近傍のサンプリングによる誘導型学習 | 大規模グラフ |
| GAT（Graph Attention Network）| アテンション重みによる近傍集約 | 不均一グラフ |
| GIN（Graph Isomorphism Network）| WL テスト同等の表現力 | グラフ分類 |
| MPNN（Message Passing NN）| メッセージパッシングの一般化フレームワーク | 分子特性予測 |

## グラフデータの表現

グラフ `G = (V, E)` はノード集合 V とエッジ集合 E からなる。

```python
import numpy as np
import torch

# --- 隣接行列による表現 ---
# 4ノードのグラフ（無向）
adj_matrix = np.array([
    [0, 1, 1, 0],
    [1, 0, 1, 1],
    [1, 1, 0, 0],
    [0, 1, 0, 0],
])

# --- エッジリスト（COO 形式）による表現 ---
# (source, target) のペア
edge_index = torch.tensor([
    [0, 1, 0, 2, 1, 2, 1, 3],   # source ノード
    [1, 0, 2, 0, 2, 1, 3, 1],   # target ノード
], dtype=torch.long)

# --- ノード特徴行列 ---
num_nodes   = 4
feature_dim = 8
x = torch.randn(num_nodes, feature_dim)   # shape: (N, F)

print("ノード数:", num_nodes)
print("エッジ数:", edge_index.shape[1] // 2, "（無向グラフ）")
print("特徴次元:", feature_dim)

# --- 正規化済み隣接行列（GCN で使用）---
def normalize_adjacency(A: np.ndarray) -> np.ndarray:
    """D^{-1/2} A D^{-1/2} の計算"""
    A_hat = A + np.eye(A.shape[0])          # 自己ループを追加
    D     = np.diag(A_hat.sum(axis=1))
    D_inv_sqrt = np.diag(1.0 / np.sqrt(np.diag(D)))
    return D_inv_sqrt @ A_hat @ D_inv_sqrt

A_norm = normalize_adjacency(adj_matrix)
print("\n正規化済み隣接行列:")
print(np.round(A_norm, 3))
```

## メッセージパッシング

GNN の中核はメッセージパッシング機構。各ノードは近傍ノードからメッセージを受け取り自身の表現を更新する。

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class MessagePassingLayer(nn.Module):
    """シンプルなメッセージパッシング層（numpy ベース）"""

    def __init__(self, in_dim: int, out_dim: int):
        super().__init__()
        self.W_self = nn.Linear(in_dim, out_dim, bias=False)
        self.W_neigh = nn.Linear(in_dim, out_dim, bias=False)
        self.bias = nn.Parameter(torch.zeros(out_dim))

    def forward(self, x: torch.Tensor,
                adj: torch.Tensor) -> torch.Tensor:
        """
        x   : (N, in_dim)  ノード特徴
        adj : (N, N)       正規化済み隣接行列
        """
        # 近傍からのメッセージ集約（平均）
        agg_neigh  = adj @ x                       # (N, in_dim)
        # 自分自身の情報 + 近傍情報の変換
        h = self.W_self(x) + self.W_neigh(agg_neigh) + self.bias
        return F.relu(h)                           # (N, out_dim)
```

## GCN（Graph Convolutional Network）

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class GCNLayer(nn.Module):
    """Kipf & Welling (2017) の GCN 層"""

    def __init__(self, in_features: int, out_features: int):
        super().__init__()
        self.linear = nn.Linear(in_features, out_features, bias=False)
        nn.init.xavier_uniform_(self.linear.weight)

    def forward(self, x: torch.Tensor, adj_norm: torch.Tensor) -> torch.Tensor:
        # H^{(l+1)} = σ(Ã H^{(l)} W^{(l)})
        support = self.linear(x)            # (N, out)
        return F.relu(adj_norm @ support)   # (N, out)


class GCN(nn.Module):
    """2 層 GCN（ノード分類）"""

    def __init__(self, in_features: int, hidden_dim: int, num_classes: int,
                 dropout: float = 0.5):
        super().__init__()
        self.gc1     = GCNLayer(in_features, hidden_dim)
        self.gc2     = GCNLayer(hidden_dim,  num_classes)
        self.dropout = dropout

    def forward(self, x: torch.Tensor, adj_norm: torch.Tensor) -> torch.Tensor:
        x = self.gc1(x, adj_norm)
        x = F.dropout(x, p=self.dropout, training=self.training)
        x = self.gc2(x, adj_norm)
        return F.log_softmax(x, dim=1)   # ノード分類用


# --- 動作確認 ---
N = 10          # ノード数
F_in = 16       # 入力特徴次元
n_class = 4     # クラス数

# ランダムグラフ
adj = torch.rand(N, N)
adj = ((adj + adj.T) / 2 > 0.5).float()  # 対称化
# 正規化
D_inv_sqrt = torch.diag(adj.sum(1).pow(-0.5))
adj_norm = D_inv_sqrt @ adj @ D_inv_sqrt

x_feat = torch.randn(N, F_in)

gcn = GCN(in_features=F_in, hidden_dim=32, num_classes=n_class)
out = gcn(x_feat, adj_norm)
print("GCN 出力 shape:", out.shape)   # (10, 4)
```

## PyTorch Geometric による実装

PyTorch Geometric（PyG）は GNN 実装の標準的ライブラリ。

```python
# pip install torch-geometric

from torch_geometric.nn import GCNConv, GATConv, global_mean_pool
from torch_geometric.data import Data, DataLoader as GeoLoader
import torch
import torch.nn as nn
import torch.nn.functional as F

# --- ノード分類モデル ---
class GCNNodeClassifier(nn.Module):
    def __init__(self, in_channels: int, hidden_channels: int,
                 out_channels: int):
        super().__init__()
        self.conv1 = GCNConv(in_channels, hidden_channels)
        self.conv2 = GCNConv(hidden_channels, out_channels)

    def forward(self, x, edge_index):
        x = F.relu(self.conv1(x, edge_index))
        x = F.dropout(x, p=0.5, training=self.training)
        return F.log_softmax(self.conv2(x, edge_index), dim=1)


# --- グラフ分類モデル（分子特性予測など）---
class GCNGraphClassifier(nn.Module):
    def __init__(self, in_channels: int, hidden_channels: int,
                 num_classes: int):
        super().__init__()
        self.conv1 = GCNConv(in_channels, hidden_channels)
        self.conv2 = GCNConv(hidden_channels, hidden_channels)
        self.conv3 = GCNConv(hidden_channels, hidden_channels)
        self.lin   = nn.Linear(hidden_channels, num_classes)

    def forward(self, x, edge_index, batch):
        # ノード埋め込みの更新
        x = F.relu(self.conv1(x, edge_index))
        x = F.relu(self.conv2(x, edge_index))
        x = F.relu(self.conv3(x, edge_index))
        # グラフ全体の表現（グローバルプーリング）
        x = global_mean_pool(x, batch)     # (グラフ数, hidden)
        return self.lin(x)


# データオブジェクトの作成
x = torch.randn(6, 9)        # 6 ノード、特徴次元 9
edge_index = torch.tensor([
    [0, 1, 1, 2, 2, 3, 3, 4, 4, 5],
    [1, 0, 2, 1, 3, 2, 4, 3, 5, 4],
], dtype=torch.long)
y = torch.tensor([0])         # グラフのラベル

data = Data(x=x, edge_index=edge_index, y=y)
print(data)
# Data(x=[6, 9], edge_index=[2, 10], y=[1])
```

## 主な応用

| 分野 | タスク | 手法 |
|------|--------|------|
| 創薬・分子設計 | 薬効予測・分子生成 | MPNN / SchNet |
| ソーシャルネットワーク | ユーザー分類・リンク予測 | GraphSAGE / GAT |
| 知識グラフ | エンティティ・リレーション推論 | R-GCN / RotatE |
| 交通予測 | 速度・渋滞予測 | ST-GCN |
| 異常検知 | 不正取引検知 | GNN + 対照学習 |
