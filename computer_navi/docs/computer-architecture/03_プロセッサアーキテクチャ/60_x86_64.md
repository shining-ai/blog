# x86-64 アセンブリ入門 (x86-64 Assembly)

## x86-64 とは

x86-64 とは、

> AMD が設計し Intel が採用した 64 ビット拡張命令セットアーキテクチャ（AMD64）で、現代の PC・サーバ・クラウド環境で主流の ISA

です。<br/>

x86-64 は IA-32（32 ビット x86）との後方互換性を保ちつつ、64 ビットアドレス空間（理論値 16 EB）・16 本の汎用レジスタ・SSE2 標準搭載を実現しています。Linux・Windows・macOS いずれも x86-64 をネイティブにサポートしています。

## 汎用レジスタ一覧

| 64 ビット | 32 ビット | 16 ビット | 8 ビット（低） | System V AMD64 ABI での用途 |
| --- | --- | --- | --- | --- |
| rax | eax | ax | al | 戻り値・アキュムレータ |
| rbx | ebx | bx | bl | 汎用（callee-saved） |
| rcx | ecx | cx | cl | 第 4 引数・カウンタ |
| rdx | edx | dx | dl | 第 3 引数・mul/div |
| rsi | esi | si | sil | 第 2 引数・ソースインデックス |
| rdi | edi | di | dil | 第 1 引数・デスティネーション |
| rsp | esp | sp | spl | スタックポインタ（callee-saved） |
| rbp | ebp | bp | bpl | フレームポインタ（callee-saved） |
| r8 | r8d | r8w | r8b | 第 5 引数 |
| r9 | r9d | r9w | r9b | 第 6 引数 |
| r10〜r11 | r10d〜r11d | r10w〜r11w | r10b〜r11b | 一時レジスタ（caller-saved） |
| r12〜r15 | r12d〜r15d | r12w〜r15w | r12b〜r15b | 汎用（callee-saved） |

## 基本命令一覧

| 命令 | 構文例 | 動作 |
| --- | --- | --- |
| mov | `mov rax, rbx` | rax = rbx（コピー） |
| add | `add rax, 5` | rax += 5 |
| sub | `sub rsp, 8` | rsp -= 8 |
| imul | `imul rax, rbx` | rax = rax * rbx（符号付き） |
| idiv | `idiv rcx` | rax = rdx:rax / rcx, rdx = 余り |
| xor | `xor rax, rax` | rax = 0（レジスタのゼロクリア） |
| push | `push rax` | rsp -= 8; [rsp] = rax |
| pop | `pop rbx` | rbx = [rsp]; rsp += 8 |
| jmp | `jmp label` | 無条件ジャンプ |
| jz | `jz label` | ZF=1 なら（比較結果が等しければ）ジャンプ |
| jnz | `jnz label` | ZF=0 ならジャンプ |
| jl | `jl label` | SF≠OF なら（符号付き小なら）ジャンプ |
| cmp | `cmp rax, rbx` | rax-rbx の結果でフラグ更新（rax は変更しない） |
| lea | `lea rax, [rbx+8]` | rax = rbx + 8（アドレス計算） |
| call | `call func` | push rip; jmp func |
| ret | `ret` | pop rip（呼び出し元に戻る） |

## メモリアドレッシングモード

```
構文: [base + index * scale + displacement]

例:
  [rax]              = *rax
  [rax + 8]          = *(rax + 8)          ← 構造体フィールドアクセス
  [rbx + rcx*4]      = *(rbx + rcx*4)      ← int 配列のインデックスアクセス
  [rbx + rcx*8 + 16] = *(rbx + rcx*8 + 16) ← long 配列 + オフセット

制約:
  base:  任意の 64 ビットレジスタ
  index: rsp 以外の任意の 64 ビットレジスタ
  scale: 1, 2, 4, 8
  disp:  8 ビットまたは 32 ビット即値
```

## 関数呼び出し規約（System V AMD64 ABI）

```
引数の渡し方（整数/ポインタ）:
  第1引数: rdi
  第2引数: rsi
  第3引数: rdx
  第4引数: rcx
  第5引数: r8
  第6引数: r9
  第7引数以降: スタック（右から左の順）

戻り値: rax（整数/ポインタ）、rdx:rax（128ビット）

callee-saved（関数内で保存が必要）: rbx, rbp, r12〜r15
caller-saved（呼び出し前後で変化しうる）: rax, rcx, rdx, rsi, rdi, r8〜r11
```

## 実装

```c title="インラインアセンブリと C 関数の対比（C）"
#include <stdio.h>
#include <stdint.h>

/* --- C で書いた関数 --- */
int64_t c_add(int64_t a, int64_t b) {
    return a + b;
}

int64_t c_popcount(uint64_t x) {
    int count = 0;
    while (x) { count += x & 1; x >>= 1; }
    return count;
}

/* --- インラインアセンブリ版 --- */
int64_t asm_add(int64_t a, int64_t b) {
    int64_t result;
    __asm__ volatile (
        "add %1, %2\n\t"      /* %2 (a) += %1 (b) */
        "mov %2, %0\n\t"      /* result = %2 */
        : "=r" (result)       /* 出力オペランド */
        : "r"  (b), "r" (a)  /* 入力オペランド */
        : /* clobbers: なし */
    );
    return result;
}

/* POPCNT 命令を直接使用（SSE4.2）*/
int64_t asm_popcnt(uint64_t x) {
    uint64_t count;
    __asm__ volatile (
        "popcnt %1, %0"
        : "=r" (count)
        : "r"  (x)
    );
    return (int64_t)count;
}

/* LEA で複数の算術演算を 1 命令で行う */
int64_t asm_lea_mul5(int64_t x) {
    int64_t result;
    /* lea rax, [rdi + rdi*4]  →  rax = x + x*4 = 5*x */
    __asm__ volatile (
        "lea (%1,%1,4), %0"
        : "=r" (result)
        : "r"  (x)
    );
    return result;
}

/* ループのアセンブリ: 1〜n の総和 */
int64_t asm_sum(int64_t n) {
    int64_t result;
    __asm__ volatile (
        "xor  %0, %0\n\t"        /* result = 0 */
        "test %1, %1\n\t"        /* n <= 0? */
        "jle  2f\n\t"            /* ならスキップ */
        "1:\n\t"
        "add  %1, %0\n\t"        /* result += n */
        "dec  %1\n\t"            /* n-- */
        "jnz  1b\n\t"            /* n != 0 なら繰り返し */
        "2:"
        : "=r" (result), "+r" (n)
        :
        : "cc"  /* フラグレジスタを変更 */
    );
    return result;
}

int main(void) {
    printf("C add(3,4)    = %ld\n", c_add(3, 4));
    printf("ASM add(3,4)  = %ld\n", asm_add(3, 4));
    printf("popcount(255) = %ld\n", asm_popcnt(255));
    printf("lea_mul5(7)   = %ld\n", asm_lea_mul5(7));
    printf("sum(100)      = %ld\n", asm_sum(100));
    return 0;
}
```

```python title="x86-64 アセンブリ出力確認と機械語実行（Python）"
import subprocess
import tempfile
import os

# --- gcc でコンパイルしたアセンブリを確認 ---
def get_asm(c_source: str, func_name: str = "") -> str:
    """C ソースをコンパイルしてアセンブリ出力を取得"""
    with tempfile.NamedTemporaryFile(suffix=".c", mode="w", delete=False) as f:
        f.write(c_source)
        src = f.name
    try:
        result = subprocess.run(
            ["gcc", "-O2", "-S", "-o", "-", src],
            capture_output=True, text=True
        )
        if result.returncode != 0:
            return f"コンパイルエラー:\n{result.stderr}"
        asm = result.stdout
        if func_name:
            # 関数のみ抽出
            lines, capture = [], False
            for line in asm.splitlines():
                if f"{func_name}:" in line: capture = True
                if capture:
                    lines.append(line)
                    if line.strip() == "ret": break
            return "\n".join(lines)
        return asm
    finally:
        os.unlink(src)


# C ソース: 単純な加算・ループ
c_add_src = """
long add(long a, long b) { return a + b; }
long sum(long n) {
    long s = 0;
    for (long i = 1; i <= n; i++) s += i;
    return s;
}
"""

print("=== add() のアセンブリ（-O2）===")
print(get_asm(c_add_src, "add"))

print("\n=== sum() のアセンブリ（-O2）===")
print(get_asm(c_add_src, "sum"))

# --- x86-64 命令の手動デコード（教育目的）---
def decode_modrm(byte: int) -> tuple[int, int, int]:
    """ModRM バイトを (mod, reg, r/m) に分解"""
    mod = (byte >> 6) & 0x3
    reg = (byte >> 3) & 0x7
    rm  =  byte       & 0x7
    return mod, reg, rm

# REX.W + MOV reg, imm64 の先頭バイトを解析
sample_bytes = [0x48, 0xB8, 0x01, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00]
print("\n=== 機械語バイトの手動解析 ===")
print(f"バイト列: {[hex(b) for b in sample_bytes]}")
print(f"  0x48 = REX プレフィックス (REX.W=1: 64ビット演算)")
print(f"  0xB8 = MOV rax, imm64 (B8+rd = MOV r64, imm64, rd=0 → rax)")
imm = int.from_bytes(sample_bytes[2:], 'little')
print(f"  即値 = {imm} (0x{imm:016X})")
```

## 使用場面

- **OS 開発**: ブートローダ・カーネルのコンテキストスイッチ・例外ハンドラに直接アセンブリを使用
- **コンパイラバックエンド**: GCC/LLVM の x86-64 コードジェネレータによる機械語生成
- **セキュリティ（ROP）**: Return-Oriented Programming 攻撃の理解と防御策（ASLR・CFG）
- **パフォーマンス最適化**: `perf annotate` でホットスポットのアセンブリを確認し手動最適化
- **リバースエンジニアリング**: IDA Pro・Ghidra によるバイナリ解析
